    public static void Serialize(in T value, out ReadOnlySpan<byte> serializedData){
        writer.Clear();

        serializedData = writer.WrittenSpan;
    }
    public static T Deserialize(ReadOnlySpan<byte> bytes) => MemoryPackSerializer.Deserialize<T>(bytes)!;
}
